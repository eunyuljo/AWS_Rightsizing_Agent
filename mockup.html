SELECT i.instance_id, i.instance_type, i.tags->>'Name' AS name, sg.group_name, sub.cidr_block, i.public_ip_address FROM aws_ec2_instance i JOIN aws_vpc_security_group sg ON sg.group_id = ANY(i.security_group_ids) JOIN aws_vpc_subnet sub ON sub.subnet_id = i.subnet_id WHERE sub.map_public_ip_on_launch = true AND sg.ingress_cidr_blocks @> '["0.0.0.0/0"]' AND i.instance_state = 'running' ORDER BY i.instance_type;
