SELECT i.instance_id, i.tags->>'Name' AS name, i.instance_type, ROUND(avg_cpu.avg_value::numeric, 1) AS cpu_avg_pct, ROUND(p.price_per_hour * 720, 2) AS monthly_cost FROM aws_ec2_instance i JOIN aws_cloudwatch_metric_statistic_data_point avg_cpu ON avg_cpu.metric_name = 'CPUUtilization' AND avg_cpu.dimension_value = i.instance_id AND avg_cpu.period = 86400 JOIN aws_pricing_product p ON p.instance_type = i.instance_type WHERE i.instance_state = 'running' AND avg_cpu.avg_value < 20 AND avg_cpu.timestamp >= NOW() - INTERVAL '30 days' ORDER BY monthly_cost DESC;
